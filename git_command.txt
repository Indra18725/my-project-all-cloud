# HTTPS, per-repo credential path (quickest)    this must be done other wise you will get 403  use API token password

# set credential for local repo
git config --local credential.helper ""       ## ignore the global manager  
git config --local --add credential.helper "store --file=.git/my-credentials"    #the line sets the local file
git config --local --get-all credential.helper     # verify it 


#  push to save the credentials  
git push
